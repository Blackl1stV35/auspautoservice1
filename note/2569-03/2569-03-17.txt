    3. NEW FILE: src/nlp_engine.py
    This module separates your Natural Language Processing logic from your database and webhook logic, keeping the project highly organized.

    Python
    import os
    import json
    import requests
    from groq import Groq
    from src.config import logger, GROQ_API_KEY

    # Initialize Groq for Speech-to-Text
    groq_client = Groq(api_key=GROQ_API_KEY) if GROQ_API_KEY else None

    def transcribe_audio(audio_path: str) -> str:
        """Transcribes Thai audio using Groq's Whisper API."""
        if not groq_client:
            logger.error("Groq API Key is missing. Cannot transcribe audio.")
            return ""
            
        try:
            with open(audio_path, "rb") as file:
                transcription = groq_client.audio.transcriptions.create(
                    file=(audio_path, file.read()),
                    model="whisper-large-v3",
                    prompt="Transcribe Thai language accurately, including factory terms like ทินเนอร์, สีโป๊ว, กิ๊บ.",
                    response_format="text",
                    language="th"
                )
            return transcription
        except Exception as e:
            logger.error(f"Groq STT Error: {e}")
            return ""

    def parse_requisition_text(transcript: str) -> dict:
        """Uses local Ollama (Typhoon 3B) to extract structured JSON from Thai text."""
        system_prompt = """
        You are a data extraction assistant for a Thai auto body shop. 
        Extract the employee name, material name, and quantity from the user's text.
        Return strictly a JSON object with keys: "employee_name", "item_name", "quantity".
        Do not include markdown formatting or any other text.
        """
        
        payload = {
            "model": "scb10x/llama3.2-typhoon2-3b-instruct",
            "prompt": f"{system_prompt}\n\nUser input: {transcript}",
            "stream": False,
            "format": "json" # Forces JSON mode
        }
        
        try:
            # Call local Ollama API
            response = requests.post("http://localhost:11434/api/generate", json=payload)
            response.raise_for_status()
            result = response.json()
            
            return json.loads(result["response"])
        except Exception as e:
            logger.error(f"Ollama Parsing Error: {e}")
            return {}
    4. Updated src/bot.py
    This script now fully handles the LINE Messaging API. It uses BackgroundTasks so that when an audio file takes a few seconds to process, FastAPI immediately tells LINE "Message Received," preventing the LINE bot from showing a timeout error to the user.

    Python
    import os
    import requests
    from fastapi import FastAPI, Request, BackgroundTasks, HTTPException
    from src.config import logger, LINE_CHANNEL_ACCESS_TOKEN, supabase
    from src.nlp_engine import transcribe_audio, parse_requisition_text
    from src.db_operations import get_or_create_employee, get_or_create_material
    from src.ai_engine import run_ai_analysis
    from src.etl_pipeline import run_etl_pipeline

    app = FastAPI(title="Auto Shop AI API")

    LINE_HEADERS = {
        "Authorization": f"Bearer {LINE_CHANNEL_ACCESS_TOKEN}",
        "Content-Type": "application/json"
    }

    def reply_to_line(reply_token: str, text: str):
        """Sends a reply message back to the LINE user."""
        payload = {
            "replyToken": reply_token,
            "messages": [{"type": "text", "text": text}]
        }
        requests.post("https://api.line.me/v2/bot/message/reply", headers=LINE_HEADERS, json=payload)

    def process_line_message(message_id: str, message_type: str, text: str, reply_token: str):
        """Background task to handle audio download, NLP parsing, and DB insertion."""
        transcript = text
        
        # 1. Handle Audio (Voice Memos)
        if message_type == "audio":
            audio_url = f"https://api-data.line.me/v2/bot/message/{message_id}/content"
            res = requests.get(audio_url, headers={"Authorization": f"Bearer {LINE_CHANNEL_ACCESS_TOKEN}"})
            
            audio_path = f"temp_{message_id}.m4a"
            with open(audio_path, 'wb') as f:
                f.write(res.content)
            
            transcript = transcribe_audio(audio_path)
            
            # Cleanup temp file
            if os.path.exists(audio_path):
                os.remove(audio_path)
                
        if not transcript:
            reply_to_line(reply_token, "❌ ไม่สามารถฟังเสียงได้ชัดเจน รบกวนพิมพ์ข้อความครับ")
            return
            
        # 2. Extract Data via LLM
        parsed_data = parse_requisition_text(transcript)
        if not parsed_data or "employee_name" not in parsed_data:
            reply_to_line(reply_token, "❌ AI ไม่สามารถแยกข้อมูลได้ รบกวนพิมพ์ใหม่อีกครั้งครับ")
            return
            
        # 3. Save to Database
        try:
            emp_name = parsed_data.get("employee_name", "Unknown")
            item_name = parsed_data.get("item_name", "Unknown")
            qty = float(parsed_data.get("quantity", 0))

            emp_id = get_or_create_employee(emp_name)
            mat_id = get_or_create_material(item_name)
            
            supabase.table("transactions").insert({
                "employee_id": emp_id,
                "material_id": mat_id,
                "quantity": qty
            }).execute()
            
            success_msg = f"✅ บันทึกสำเร็จ:\nพนักงาน: {emp_name}\nเบิก: {item_name}\nจำนวน: {qty}"
            reply_to_line(reply_token, success_msg)
            logger.info(f"LINE Transaction saved: {emp_name} took {qty} of {item_name}")
            
        except Exception as e:
            logger.error(f"DB Insert Error from LINE: {e}")
            reply_to_line(reply_token, "❌ เกิดข้อผิดพลาดในการบันทึกลงฐานข้อมูล")

    @app.post("/webhook")
    async def line_webhook(request: Request, background_tasks: BackgroundTasks):
        """Receives events from LINE Platform."""
        body = await request.json()
        
        for event in body.get("events", []):
            if event.get("type") == "message":
                reply_token = event["replyToken"]
                msg = event["message"]
                msg_type = msg["type"]
                msg_id = msg["id"]
                text = msg.get("text", "")
                
                # Route text and audio to the background processor
                if msg_type in ["text", "audio"]:
                    background_tasks.add_task(process_line_message, msg_id, msg_type, text, reply_token)
                    
        return {"status": "ok"}

    @app.post("/trigger-etl")
    def api_trigger_etl():
        """Endpoint to trigger the ETL process manually"""

        try:
            run_etl_pipeline(req_file, pur_file)
            run_ai_analysis()
            return {"status": "Success", "message": "Pipeline and AI analysis executed."}

Your setup_metabase.py script is correctly trying to use this shared pooler (aws-0-ap-southeast-1.pooler.supabase.com). The problem is that Supabase's recent infrastructure updates have made third-party BI tools (like Metabase) incredibly finicky when authenticating through that shared pooler, resulting in the dreaded "Tenant or user not found" error you experienced earlier.
